== 1. halaman register tampil ==
GET /register -> 200
   ok: judul Daftar ada
== 2. email duplikat ditolak ==
POST /register (admin@queuecare.id) -> 200 
   ok: pesan email duplikat ada
== 3. registrasi baru ==
POST /register (baru@queuecare.id) -> 302 http://localhost:8081/login
GET /login -> 200
   ok: flash sukses tampil
== 4. password tidak sama ==
POST /register (beda) -> 200 
   ok: pesan konfirmasi tampil
== 5. login salah password ==
POST /login (salah) -> 302 http://localhost:8081/login?error
GET /login?error -> 200
   ok: pesan salah tampil
== 6. login customer ==
POST /login -> 302 http://localhost:8081/customer
GET /customer -> 200
   ok: halaman customer
GET /admin (customer) -> 403
   ok: 403 + halaman akses ditolak
GET /owner (customer) -> 403
   ok: 403 + halaman akses ditolak
== 7. logout ==
POST /logout -> 302 http://localhost:8081/login?logout
GET /customer (sudah keluar) -> 302
GET /login?logout -> 200
   ok: pesan logout tampil
== 8. login admin ==
